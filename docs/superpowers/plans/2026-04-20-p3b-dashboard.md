# P3b — Read-Only Dashboard + SSE Live-Tail Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a read-only React dashboard (Overview, Repos, Schedules, Runs, Run Detail with SSE live-tail) embedded in the Go binary and secured by the P3a session layer.

**Architecture:** Vite+React SPA in `/web`, built to `web/dist/` and embedded via `embed.FS` in the Go binary. Six new `/api/*` endpoints in `internal/webapi` return JSON for the five pages. The SSE endpoint streams `run_event` rows for in-flight runs, closing on terminal status.

**Tech Stack:** Go 1.22+, React 18, TypeScript, Vite, React Router v6, TanStack Query v5, shadcn/ui, Radix UI, Tailwind CSS, pgx/v5.

-- name: GetOverviewStats :one
SELECT
  COUNT(*) FILTER (WHERE enabled = true)  AS schedules_enabled,
  COUNT(*)                                 AS schedules_total,
  (
    SELECT COUNT(*) FROM run
    WHERE org_id = $1
      AND created_at >= now() - interval '24 hours'
  ) AS runs_today,
  (
    SELECT COUNT(*) FROM run
    WHERE org_id = $1
      AND status IN ('failed','partial_failure')
      AND created_at >= now() - interval '24 hours'
  ) AS failures_last_24h,
  (
    SELECT MAX(started_at) FROM run
    WHERE org_id = $1
  ) AS last_run_at
FROM schedule
WHERE org_id = $1;

-- name: GetCostStats :one
SELECT
  COALESCE(SUM(tokens_in + tokens_out) FILTER (
    WHERE created_at >= date_trunc('day', now())
  ), 0)::bigint AS tokens_today,
  COALESCE(SUM(cost_cents) FILTER (
    WHERE created_at >= date_trunc('week', now())
  ), 0)::bigint AS cost_this_week_cents,
  COALESCE(SUM(cost_cents) FILTER (
    WHERE created_at >= date_trunc('month', now())
  ), 0)::bigint AS cost_this_month_cents
FROM run
WHERE org_id = $1;

-- name: ListRecentRuns :many
SELECT r.id, r.status, r.fire_reason, r.started_at, r.finished_at,
       r.duration_ms, r.tokens_in, r.tokens_out, r.cost_cents,
       r.error_kind, r.error_msg,
       s.name AS schedule_name,
       sk.path AS skill_path,
       rc.owner, rc.name AS repo_name

JOIN skill sk           ON sk.id = s.skill_id
JOIN repo_connection rc ON rc.id = sk.repo_id
WHERE r.org_id = $1
ORDER BY r.created_at DESC
LIMIT $2 OFFSET $3;

package webapi

import (
	"net/http"

	"github.com/jackc/pgx/v5/pgxpool"
)

// Deps holds everything webapi handlers need.
type Deps struct {
	MasterKey         []byte
	OAuthClientID     string
	OAuthClientSecret string
	AdminLogins       []string
	ViewerLogins      []string
	Pool              *pgxpool.Pool
	// GitHubAPIBase overrides the GitHub API base URL in tests. Empty = real GitHub.
	GitHubAPIBase string
}

// resolveRole returns "admin", "viewer", or "" (not allowed).
func (d Deps) resolveRole(login string) string {
	for _, l := range d.AdminLogins {
		if l == login {
			return "admin"
		}
	}
	for _, l := range d.ViewerLogins {
		if l == login {
			return "viewer"
		}
	}
	return ""
}

// RegisterRoutes registers /oauth/*, /api/* on mux, and the static SPA handler.
func RegisterRoutes(mux *http.ServeMux, deps Deps) {
	session := func(h http.Handler) http.Handler {
		return RequireSession(deps.MasterKey, h)
	}

	// Existing
	mux.Handle("GET /api/me", session(meHandler{}))
	oh := oauthHandlers{deps: deps}
	mux.HandleFunc("GET /oauth/login", oh.login)
	mux.HandleFunc("GET /oauth/callback", oh.callback)
	mux.HandleFunc("GET /oauth/logout", oh.logout)

	// P3b read endpoints
	mux.Handle("GET /api/overview", session(overviewHandler{pool: deps.Pool}))
	mux.Handle("GET /api/repos", session(reposHandler{pool: deps.Pool}))
	mux.Handle("GET /api/schedules", session(schedulesHandler{pool: deps.Pool}))
	mux.Handle("GET /api/runs", session(runsHandler{pool: deps.Pool}))
	mux.Handle("GET /api/runs/{id}", session(runDetailHandler{pool: deps.Pool}))
	mux.Handle("GET /api/runs/{id}/events", session(runEventsSSEHandler{pool: deps.Pool}))
}
```

package webapi

import (
	"encoding/json"
	"net/http"

	"github.com/jackc/pgx/v5/pgxpool"

	dbgen "github.com/gambtho/cronfoundry/internal/db/gen"

type overviewHandler struct{ pool *pgxpool.Pool }

type overviewResponse struct {
	Health overviewHealth `json:"health"`
	Cost   overviewCost   `json:"cost"`
}

type overviewHealth struct {
	SchedulesEnabled int    `json:"schedulesEnabled"`
	SchedulesTotal   int    `json:"schedulesTotal"`
	RunsToday        int64  `json:"runsToday"`
	FailuresLast24h  int64  `json:"failuresLast24h"`
	LastRunAt        *int64 `json:"lastRunAt"` // Unix seconds, nil if no runs
}

type overviewCost struct {
	TokensToday        int64 `json:"tokensToday"`
	CostThisWeekCents  int64 `json:"costThisWeekCents"`
	CostThisMonthCents int64 `json:"costThisMonthCents"`
}

func (h overviewHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
	claims := SessionClaimsFromContext(r.Context())
	orgID := mustOrgID(claims)

	q := dbgen.New(h.pool)
	health, err := q.GetOverviewStats(r.Context(), orgID)
	if err != nil {
		http.Error(w, "internal error", http.StatusInternalServerError)
		return
	}
	cost, err := q.GetCostStats(r.Context(), orgID)
	if err != nil {
		http.Error(w, "internal error", http.StatusInternalServerError)
		return
	}

	var lastRunAt *int64
	if health.LastRunAt.Valid {
		t := health.LastRunAt.Time.Unix()
		lastRunAt = &t
	}

	resp := overviewResponse{
		Health: overviewHealth{
			SchedulesEnabled: int(health.SchedulesEnabled.(int64)),
			SchedulesTotal:   int(health.SchedulesTotal.(int64)),
			RunsToday:        health.RunsToday.(int64),
			FailuresLast24h:  health.FailuresLast24h.(int64),
			LastRunAt:        lastRunAt,
		},
		Cost: overviewCost{

			CostThisWeekCents:  cost.CostThisWeekCents,
			CostThisMonthCents: cost.CostThisMonthCents,
		},
	}

	w.Header().Set("Content-Type", "application/json")
	_ = json.NewEncoder(w).Encode(resp)
}

import (
	"encoding/json"
	"errors"
	"fmt"
	"net/http"
	"time"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgtype"
	"github.com/jackc/pgx/v5/pgxpool"

	dbgen "github.com/gambtho/cronfoundry/internal/db/gen"
)

type runEventsSSEHandler struct {
	pool  *pgxpool.Pool
	orgID pgtype.UUID
}

var terminalStatuses = map[string]bool{
	"succeeded":       true,
	"failed":          true,
	"partial_failure": true,
}

func (h runEventsSSEHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
	idStr := r.PathValue("id")
	var runID pgtype.UUID
	if err := runID.Scan(idStr); err != nil {
		http.Error(w, "invalid run id", http.StatusBadRequest)
		return
	}

	q := dbgen.New(h.pool)

	// Verify run exists and belongs to org
	status, err := q.GetRunStatus(r.Context(), dbgen.GetRunStatusParams{ID: runID, OrgID: h.orgID})
	if err != nil {
		if errors.Is(err, pgx.ErrNoRows) {
			http.Error(w, "not found", http.StatusNotFound)
			return
		}
		http.Error(w, "internal error", http.StatusInternalServerError)
		return
	}

	// Set SSE headers
	w.Header().Set("Content-Type", "text/event-stream")
	w.Header().Set("Cache-Control", "no-cache")
	w.Header().Set("Connection", "keep-alive")
	w.Header().Set("X-Accel-Buffering", "no")

	flusher, ok := w.(http.Flusher)
	if !ok {
		http.Error(w, "streaming not supported", http.StatusInternalServerError)
		return
	}

	// Flush existing events
	existing, err := q.ListRunEvents(r.Context(), runID)
	if err != nil {
		return
	}
	var lastID int64
	for _, ev := range existing {
		writeSSEEvent(w, ev)
		lastID = ev.ID
	}
	flusher.Flush()

	// If already terminal, close immediately
	if terminalStatuses[status] {
		fmt.Fprintf(w, "event: done\ndata: {}\n\n")
		flusher.Flush()
		return
	}

	// Poll for new events
	ticker := time.NewTicker(500 * time.Millisecond)
	defer ticker.Stop()

	for {
		select {
		case <-r.Context().Done():
			return
		case <-ticker.C:
			newEvents, err := q.ListRunEventsSince(r.Context(), dbgen.ListRunEventsSinceParams{
				RunID:      runID,
				LastSeenID: lastID,
			})
			if err != nil {
				return
			}
			for _, ev := range newEvents {
				writeSSEEvent(w, ev)
				lastID = ev.ID
			}
			if len(newEvents) > 0 {
				flusher.Flush()
			}

			// Check if run reached terminal status
			currentStatus, err := q.GetRunStatus(r.Context(), dbgen.GetRunStatusParams{ID: runID, OrgID: h.orgID})
			if err != nil {
				return
			}
			if terminalStatuses[currentStatus] {
				fmt.Fprintf(w, "event: done\ndata: {}\n\n")
				flusher.Flush()
				return
			}
		}
	}
}

func writeSSEEvent(w http.ResponseWriter, ev dbgen.RunEvent) {
	data, _ := json.Marshal(map[string]interface{}{
		"id":        ev.ID,
		"ts":        ev.Ts.Time.Unix(),
		"level":     ev.Level,
		"eventType": ev.EventType,
		"payload":   json.RawMessage(ev.PayloadJson),
	})
	fmt.Fprintf(w, "data: %s\n\n", data)
}

  const res = await fetch(path, init)
  if (res.status === 401 || res.status === 403) {
    window.location.href = '/oauth/login'

  }
  if (!res.ok) {

    if (TERMINAL.has(run.status)) {
      // Already terminal — events already in run.events, no SSE needed
      esRef.current?.close()
      return
    }

    // Open SSE for in-flight run
    const es = new EventSource(`/api/runs/${run.id}/events`)
    esRef.current = es

    es.onmessage = (e) => {
      const ev: RunEvent = JSON.parse(e.data)
      setLiveEvents((prev) => [...prev, ev])
    }

    es.addEventListener('done', () => {
      es.close()
      // Refetch run detail to get final status/duration/cost
      queryClient.invalidateQueries({ queryKey: ['run', run.id] })
    })

    es.onerror = () => es.close()

    return () => es.close()
  }, [run?.id, run?.status])

  if (isLoading) return <div className="text-muted-foreground">Loading...</div>
  if (error || !run) return <div className="text-destructive">Run not found.</div>

  const events = TERMINAL.has(run.status) ? run.events : liveEvents

  return (
    <div className="space-y-6">
      <div className="flex items-center gap-4">
        <h1 className="text-xl font-bold">{run.scheduleName}</h1>
        <StatusBadge status={run.status} />
        <span className="text-sm text-muted-foreground">{run.fireReason}</span>
      </div>

      <div className="grid grid-cols-2 md:grid-cols-4 gap-4 text-sm">
        <div><span className="text-muted-foreground">Started:</span> {run.startedAt ? new Date(run.startedAt * 1000).toLocaleString() : '—'}</div>
        <div><span className="text-muted-foreground">Duration:</span> {run.durationMs != null ? `${(run.durationMs / 1000).toFixed(1)}s` : '—'}</div>
        <div><span className="text-muted-foreground">Tokens:</span> {run.tokensIn != null ? `${run.tokensIn} in / ${run.tokensOut} out` : '—'}</div>
        <div><span className="text-muted-foreground">Cost:</span> {run.costCents != null ? `$${(run.costCents / 100).toFixed(4)}` : '—'}</div>
      </div>

      {run.errorMsg && (
        <div className="p-3 bg-destructive/10 rounded text-sm text-destructive">
          {run.errorKind}: {run.errorMsg}
        </div>
      )}

      {run.writebackCommitSha && (
        <div className="text-sm">
          <span className="text-muted-foreground">Writeback commit:</span>{' '}
          <span className="font-mono">{run.writebackCommitSha.slice(0, 8)}</span>
        </div>
      )}

      <div>
        <h2 className="text-sm font-semibold mb-2">
          Event Timeline
          {!TERMINAL.has(run.status) && (
            <span className="ml-2 text-xs text-muted-foreground animate-pulse">● live</span>
          )}
        </h2>
        <EventTimeline events={events} />
      </div>
    </div>
  )
}
```

import "embed"
import "io/fs"

//go:embed all:../../web/dist
var webDist embed.FS

func spaHandler() http.Handler {
	dist, err := fs.Sub(webDist, "web/dist")
	if err != nil {
		panic("embed: " + err.Error())
	}
	fileServer := http.FileServer(http.FS(dist))
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		// Serve file if it exists; otherwise serve index.html for SPA routing
		path := r.URL.Path
		if path == "/" || path == "" {
			r.URL.Path = "/index.html"
		}
		// Try to open the requested path
		f, err := dist.Open(strings.TrimPrefix(path, "/"))
		if err != nil {
			// Fall back to index.html
			r.URL.Path = "/index.html"
		} else {
			f.Close()
		}
		fileServer.ServeHTTP(w, r)
	})
}
