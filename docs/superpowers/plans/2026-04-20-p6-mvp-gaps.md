# Phase P6a — GitHub push-webhook resync

**Why:** The spec (§"Sync model") calls for a GitHub App **push webhook** to re-parse `cronfoundry.yaml` + `SKILL.md` and upsert skill/schedule rows with the new SHA. Today only a 60-second poller exists (`internal/sync/poller.go`), so repo changes land with up to a minute of lag.

**Approach:** Add one unauthenticated endpoint `POST /webhook/github` that verifies `X-Hub-Signature-256` (HMAC-SHA256 of the raw body with a shared secret), decodes the `push` payload, looks up the `repo_connection` by `(owner, name)`, and calls the existing `sync.Poller.SyncOne`. Reuse the existing clone-parse-upsert path — no new sync logic needed.

**Shared secret storage:** one global env var `CRONFOUNDRY_GITHUB_WEBHOOK_SECRET`, set at deploy time when the GitHub App is registered. A single secret per deployment matches the single-GitHub-App-per-deployment model in the spec.

```go
// internal/github/webhook.go
package github

import (
	"crypto/hmac"
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"strings"
)

// VerifyWebhookSignature checks the GitHub-supplied X-Hub-Signature-256 header
// against an HMAC-SHA256 of body using secret. sig must be "sha256=<hex>".
func VerifyWebhookSignature(secret, body []byte, sig string) error {
	const prefix = "sha256="
	if !strings.HasPrefix(sig, prefix) {
		return errors.New("webhook: missing sha256= prefix")
	}
	want, err := hex.DecodeString(sig[len(prefix):])
	if err != nil || len(want) == 0 {
		return errors.New("webhook: invalid hex signature")
	}
	mac := hmac.New(sha256.New, secret)
	mac.Write(body)
	got := mac.Sum(nil)
	if !hmac.Equal(got, want) {
		return errors.New("webhook: signature mismatch")
	}
	return nil
}

// PushPayload is the subset of the GitHub push event we care about.
type PushPayload struct {
	Ref        string `json:"ref"`
	After      string `json:"after"`
	Repository struct {
		Name  string `json:"name"`
		Owner struct {
			Login string `json:"login"`
		} `json:"owner"`
		DefaultBranch string `json:"default_branch"`
	} `json:"repository"`
	Installation struct {
		ID int64 `json:"id"`
	} `json:"installation"`
}
```

```go
// internal/webapi/webhook.go
package webapi

import (
	"context"
	"crypto/subtle"
	"encoding/json"
	"errors"
	"io"
	"log/slog"
	"net/http"
	"strings"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgtype"

	dbgen "github.com/gambtho/cronfoundry/internal/db/gen"
	gh "github.com/gambtho/cronfoundry/internal/github"
)

// RepoSyncer resolves a repo_connection by (org_id, owner, name) and triggers
// a single sync pass. The concrete implementation in serve.go wraps
// sync.Poller.SyncOne.
type RepoSyncer interface {
	SyncOne(ctx context.Context, connID pgtype.UUID) error
}

type webhookHandler struct {
	deps   Deps
	secret []byte
	syncer RepoSyncer
}

func (h *webhookHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
	event := r.Header.Get("X-GitHub-Event")
	// Respond 200 to ping so the UI can verify the webhook setup.
	if event == "ping" {
		w.WriteHeader(http.StatusOK)
		return
	}
	if event != "push" {
		w.WriteHeader(http.StatusNoContent)
		return
	}
	body, err := io.ReadAll(io.LimitReader(r.Body, 5*1024*1024))
	if err != nil {
		writeErr(w, http.StatusBadRequest, "read body", "bad_request")
		return
	}
	sig := r.Header.Get("X-Hub-Signature-256")
	if len(h.secret) == 0 || subtle.ConstantTimeCompare(h.secret, h.secret) != 1 /* length-safe guard */ {
		writeErr(w, http.StatusServiceUnavailable, "webhook secret not configured", "config")
		return
	}
	if err := gh.VerifyWebhookSignature(h.secret, body, sig); err != nil {
		writeErr(w, http.StatusUnauthorized, "invalid signature", "unauthorized")
		return
	}

	var p gh.PushPayload
	if err := json.Unmarshal(body, &p); err != nil {
		writeErr(w, http.StatusBadRequest, "parse payload", "bad_request")
		return
	}
	owner := p.Repository.Owner.Login
	name := p.Repository.Name
	if owner == "" || name == "" {
		writeErr(w, http.StatusBadRequest, "missing repo coords", "bad_request")
		return
	}
	// Only resync pushes to the connection's default branch. A push to
	// another branch doesn't change what schedules run.
	expectRef := "refs/heads/" + p.Repository.DefaultBranch
	if p.Repository.DefaultBranch != "" && p.Ref != expectRef {
		w.WriteHeader(http.StatusNoContent)
		return
	}

	org, err := h.deps.Queries.GetFirstOrganization(r.Context())
	if err != nil {
		writeErr(w, http.StatusInternalServerError, "load org", "internal")
		return
	}
	conn, err := h.deps.Queries.GetRepoConnectionByOwnerName(r.Context(), dbgen.GetRepoConnectionByOwnerNameParams{
		OrgID: org.ID,
		Owner: owner,
		Name:  name,
	})
	if err != nil {
		if errors.Is(err, pgx.ErrNoRows) {
			// Webhook for a repo we don't track. Respond 200 so GitHub
			// doesn't mark the hook unhealthy; log for visibility.
			slog.Info("webhook: unknown repo, ignoring", "owner", owner, "name", name)
			w.WriteHeader(http.StatusOK)
			return
		}
		writeErr(w, http.StatusInternalServerError, "lookup conn", "internal")
		return
	}

	// Sync runs synchronously. A push webhook can afford to wait a few
	// seconds; if it grows slow, push to a channel instead.
	if err := h.syncer.SyncOne(r.Context(), conn.ID); err != nil {
		slog.Warn("webhook: sync failed", "conn_id", conn.ID, "err", err)
		writeErr(w, http.StatusInternalServerError, "sync failed", "sync_error")
		return
	}
	w.WriteHeader(http.StatusOK)
}

// Helper so malformed Content-Type headers (e.g., no charset) don't trip us
// up — we only care about the JSON body anyway.
var _ = strings.EqualFold
```

# Phase P6b — Run-scoped secret-manifest logging

**Why:** The spec's "Per-run secret scoping" section requires that (i) the runner announces the allowed secret names at run start, and (ii) every KV read is logged to `run_event` keyed by `run.id`. Today `CollectSecretRefs` populates the manifest in the scheduler (`internal/scheduler/tick.go:176`) but nothing logs it, and `GET /internal/secrets` fetches without emitting events. This is the MVP surrogate for cryptographic scoping.

**Approach:** Emit three new `run_event` types:
- `manifest.set` — posted once by the runner after `GetRunContext`. Payload: `{ "allowed": ["name1","name2"] }`.
- `secret.fetched` — posted by the API's `GET /internal/secrets` handler, one event per secret name returned. Payload: `{ "name": "<name>", "in_manifest": true/false }`.
- `secret.denied` — posted by the API when a requested name is not on the schedule's manifest. (Advisory only; we still return the value in MVP per the spec — "rely on logging for MVP".)

**Dependencies:** none on P6a/c/d.

```go
// internal/audit/audit.go
// Package audit centralizes audit_log writes so call sites stay a single line.
package audit

import (
	"context"
	"encoding/json"
	"fmt"

	"github.com/google/uuid"
	"github.com/jackc/pgx/v5/pgtype"

	dbgen "github.com/gambtho/cronfoundry/internal/db/gen"
)

// Entry describes one audit event.
type Entry struct {
	OrgID      pgtype.UUID
	Actor      string // GitHub login, or "system" for scheduler-driven actions
	Action     string // e.g. "schedule.pause", "secret.rotate"
	TargetKind string // e.g. "schedule", "secret", "repo"
	TargetID   *uuid.UUID
	Detail     map[string]any
}

// Log writes one audit row. Call-site errors should NOT be fatal: audit
// writes are advisory, not mutating business state. Callers typically log
// the error and proceed.
func Log(ctx context.Context, q *dbgen.Queries, e Entry) error {
	detail := json.RawMessage(`{}`)
	if e.Detail != nil {
		b, err := json.Marshal(e.Detail)
		if err != nil {
			return fmt.Errorf("audit: marshal detail: %w", err)
		}
		detail = b
	}
	var target pgtype.UUID
	if e.TargetID != nil {
		target = pgtype.UUID{Bytes: *e.TargetID, Valid: true}
	}
	return q.InsertAuditLog(ctx, dbgen.InsertAuditLogParams{
		OrgID:       e.OrgID,
		Actor:       strPtr(e.Actor),
		Action:      e.Action,
		TargetKind:  strPtr(e.TargetKind),
		TargetID:    target,
		DetailJson:  detail,
	})
}

func strPtr(s string) *string {
	if s == "" {
		return nil
	}
	return &s
}
```

# Phase P6d — `user` table + DB-backed allowlist

**Why:** The spec's data model lists a `user` row per GitHub login, with role transitions that are audit-logged. 	if err := json.NewDecoder(r.Body).Decode(&req); err != nil || req.Login == "" || (req.Role != "admin" && req.Role != "viewer") {
```go
		writeErr(w, http.StatusBadRequest, "login + role (admin|viewer) required", "bad_request")
		return
	}
	org, _ := h.deps.Queries.GetFirstOrganization(r.Context())
	row, err := h.deps.Queries.CreateUser(r.Context(), dbgen.CreateUserParams{
		OrgID:       org.ID,
		GithubLogin: req.Login,
		Role:        req.Role,
	})
	if err != nil {
		writeErr(w, http.StatusInternalServerError, "create user", "internal")
		return
	}
	_ = audit.Log(r.Context(), h.deps.Queries, audit.Entry{
		OrgID:      org.ID,
		Actor:      mustClaims(r).Login,
		Action:     "user.create",
		TargetKind: "user",
		Detail:     map[string]any{"login": req.Login, "role": req.Role},
	})
	writeJSON(w, http.StatusCreated, row)
}

func (h *usersHandler) updateRole(w http.ResponseWriter, r *http.Request) {
	login := r.PathValue("login")
	var req struct {
		Role string `json:"role"`
	}
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil || (req.Role != "admin" && req.Role != "viewer") {
		writeErr(w, http.StatusBadRequest, "role must be admin|viewer", "bad_request")
		return
	}
	org, _ := h.deps.Queries.GetFirstOrganization(r.Context())
	if err := h.deps.Queries.SetUserRole(r.Context(), dbgen.SetUserRoleParams{
		OrgID:       org.ID,
		GithubLogin: login,
		Role:        req.Role,
	}); err != nil {
		writeErr(w, http.StatusInternalServerError, "update role", "internal")
		return
	}
	_ = audit.Log(r.Context(), h.deps.Queries, audit.Entry{
		OrgID:      org.ID,
		Actor:      mustClaims(r).Login,
		Action:     "user.set_role",
		TargetKind: "user",
		Detail:     map[string]any{"login": login, "role": req.Role},
	})
	w.WriteHeader(http.StatusNoContent)
}

func (h *usersHandler) delete(w http.ResponseWriter, r *http.Request) {
	login := r.PathValue("login")
	// Refuse to delete yourself — it would lock the caller out.
	if mustClaims(r).Login == login {
		writeErr(w, http.StatusConflict, "cannot delete your own user", "self_delete")
		return
	}
	org, _ := h.deps.Queries.GetFirstOrganization(r.Context())
	// Refuse to delete the last admin.
	count, _ := h.deps.Queries.CountUsers(r.Context(), org.ID)
	if count <= 1 {
		writeErr(w, http.StatusConflict, "cannot delete the last user", "last_user")
		return
	}
	if err := h.deps.Queries.DeleteUser(r.Context(), dbgen.DeleteUserParams{
		OrgID:       org.ID,
		GithubLogin: login,
	}); err != nil {
		writeErr(w, http.StatusInternalServerError, "delete user", "internal")
		return
	}
	_ = audit.Log(r.Context(), h.deps.Queries, audit.Entry{
		OrgID:      org.ID,
		Actor:      mustClaims(r).Login,
		Action:     "user.delete",
		TargetKind: "user",
		Detail:     map[string]any{"login": login},
	})
	w.WriteHeader(http.StatusNoContent)
}
```
